from flask import Flask, render_template, request

app = Flask(_name_)

@app.route("/", methods=["GET", "POST"])
def index():
    total = None
    pesan = None

    if request.method == "POST":
        saldo = int(request.form["saldo"])
        ukuran = request.form["ukuran"]
        extra = request.form.get("extra")

        if saldo < 100000:
            pesan = "Saldo tidak cukup"
        else:
            if ukuran == "Small":
                harga = 25000
            elif ukuran == "Medium":
                harga = 40000
            elif ukuran == "Large":
                harga = 60000

            if extra == "yes":
                harga += 10000

            total = harga

    return render_template("index.html", total=total, pesan=pesan)

app.run(debug=True)
